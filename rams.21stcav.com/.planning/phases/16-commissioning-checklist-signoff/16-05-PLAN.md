```blade
<script>
function signoffSheet(initial) {
    return {
        open: false,
        step: 1,                   // 1=preview, 2=sign, 3=done
        programmeId: initial.programmeId,
        certificationText: initial.certificationText,
        previewUrl: null,
        previewError: null,
        clientName: '',
        clientRole: '',
        clientCompany: '',
        signaturePad: null,
        isEmpty: true,
        submitting: false,
        errorMessage: null,
        finalPdfUrl: null,

        get canSubmit() {
            return this.clientName.trim().length > 0
                && this.clientRole.trim().length > 0
                && this.clientCompany.trim().length > 0
                && ! this.isEmpty;
        },

        csrf() {
            return document.querySelector('meta[name=csrf-token]').content;
        },

        async openSheet() {
            this.open = true;
            this.step = 1;
            this.previewUrl = null;
            this.previewError = null;
            this.errorMessage = null;

            // Kick off preview generation
            try {
                const res = await fetch(`/install-programmes/${this.programmeId}/commissioning/signoff/preview`, {
                    method: 'POST',
                    headers: {
                        'X-CSRF-TOKEN': this.csrf(),
                        'Accept': 'application/json',
                    },
                });
                if (! res.ok) {
                    const j = await res.json().catch(() => ({ message: 'Preview failed' }));
                    this.previewError = j.message || 'Preview failed';
                    return;
                }
                const json = await res.json();
                this.previewUrl = json.preview_url;
            } catch (e) {
                this.previewError = 'Preview request failed.';
            }

            // When step flips to 2, initialise canvas (next tick)
            this.$watch('step', (val) => {
                if (val === 2) {
                    this.$nextTick(() => this.initCanvas());
                }
            });
        },

        initCanvas() {
            // ── DPI-scaled canvas (Pitfall 2 snippet verbatim) ──────────────
            // Source: github.com/szimek/signature_pad#handling-high-dpi-screens
            const canvas = this.$refs.signatureCanvas;
            if (! canvas) return;

            const resize = () => {
                const ratio = Math.max(window.devicePixelRatio || 1, 1);
                canvas.width = canvas.offsetWidth * ratio;
                canvas.height = canvas.offsetHeight * ratio;
                canvas.getContext('2d').scale(ratio, ratio);

                // Integration hook — determined by Plan 02 DPI spike.
                // Option A (window.SignaturePad global exposed by creagia bundle):
                this.signaturePad = new window.SignaturePad(canvas, {
                    minWidth: 0.5,
                    maxWidth: 2.0,
                    penColor: 'rgb(0,0,0)',
                });
                // Option B fallback — uncomment if spike chose Option B:
                // this.signaturePad = canvas.__signaturePad;

                this.signaturePad.clear();
                this.isEmpty = true;
                this.signaturePad.addEventListener('endStroke', () => {
                    this.isEmpty = this.signaturePad.isEmpty();
                });
            };

            resize();
            this.resizeHandler = resize;
            window.addEventListener('resize', resize);
            window.addEventListener('orientationchange', resize);
        },

        clearCanvas() {
            if (this.signaturePad) {
                this.signaturePad.clear();
                this.isEmpty = true;
            }
        },

        async submitFinalise() {
            this.errorMessage = null;
            if (! this.canSubmit) return;
            this.submitting = true;

            const dataUrl = this.signaturePad.toDataURL('image/png');

            try {
                const res = await fetch(`/install-programmes/${this.programmeId}/commissioning/signoff/finalise`, {
                    method: 'POST',
                    headers: {
                        'X-CSRF-TOKEN': this.csrf(),
                        'Content-Type': 'application/json',
                        'Accept': 'application/json',
                    },
                    body: JSON.stringify({
                        client_name: this.clientName,
                        client_role: this.clientRole,
                        client_company: this.clientCompany,
                        signature_png_base64: dataUrl,
                    }),
                });

                if (! res.ok) {
                    const j = await res.json().catch(() => ({ message: 'Finalise failed' }));
                    this.errorMessage = j.message || 'Finalise failed';
                    this.submitting = false;
                    return;
                }

                const json = await res.json();
                this.finalPdfUrl = json.final_pdf_url;
                this.step = 3;
            } catch (e) {
                this.errorMessage = 'Finalise request failed.';
            } finally {
                this.submitting = false;
            }
        },
    };
}
</script>
```

```php
<?php

namespace App\Http\Controllers;

use App\Exceptions\CommissioningSignoffException;
use App\Models\InstallProgramme;
use App\Services\CommissioningSyncService;
use Illuminate\Http\JsonResponse;

/**
 * CommissioningResyncController — D-04 re-sync from programme.
 *
 * Rebuilds commissioning_items from the current install_tasks while preserving
 * existing statuses for unchanged items. Refuses to run after signoff (INST-05i
 * enforced by CommissioningSyncService).
 */
class CommissioningResyncController extends Controller
{
    public function __construct(
        private readonly CommissioningSyncService $syncService,
    ) {}

    public function resync(InstallProgramme $programme): JsonResponse
    {
        $this->authorise($programme);

        try {
            $diff = $this->syncService->resync($programme);
        } catch (CommissioningSignoffException $e) {
            return response()->json(['message' => $e->getMessage()], 422);
        }

        return response()->json([
            'diff' => $diff,
        ]);
    }

    private function authorise(InstallProgramme $programme): void
    {
        $project = $programme->project;
        abort_if(
```

```javascript
resync: { open: false, loading: false, error: null, counters: null },

async triggerResync() {
    this.resync = { open: true, loading: true, error: null, counters: null };
    try {
        const res = await fetch(`/install-programmes/${this.programmeId}/commissioning/resync`, {
            method: 'POST',
            headers: {
                'X-CSRF-TOKEN': this.csrf(),
                'Accept': 'application/json',
            },
        });
        if (! res.ok) {
            const j = await res.json().catch(() => ({ message: 'Resync failed' }));
            this.resync.loading = false;
            this.resync.error = j.message || 'Resync failed';
            return;
        }
        const json = await res.json();
        this.resync.loading = false;
        this.resync.counters = json.diff;
    } catch (e) {
        this.resync.loading = false;
        this.resync.error = 'Resync request failed.';
    }
},
```
