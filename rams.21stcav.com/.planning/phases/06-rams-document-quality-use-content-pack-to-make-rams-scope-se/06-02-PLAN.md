```php
<?php

namespace Tests\Unit;

use Tests\TestCase;

class RamsPdfScopeTest extends TestCase
{
    private function renderScope(string $scopeOfWorks): string
    {
        // Minimal data needed to render the scope section of the RAMS PDF.
        // Only the $scopeOfWorks variable is relevant to this test.
        return view('pdf.rams', [
            'scopeOfWorks'   => $scopeOfWorks,
            'project'        => ['works_description' => 'should not appear'],
            'formData'       => ['works_description' => 'should not appear either'],
            'company'        => 'Test AV Ltd',
            // Provide empty stubs for all other required variables.
            'data'           => [],
            'quote'          => [],
            'ramsRef'        => 'TEST-001',
            'revisionNumber' => '1',
            'issueDate'      => now()->format('d M Y'),
            'projectName'    => 'Test Project',
            'clientName'     => 'Test Client',
            'siteAddress'    => 'Test Site',
            'quoteRef'       => 'Q001',
            'projectManager' => '',
            'methodStatement'=> ['phases' => []],
            'hazards'        => [],
            'ppe'            => [],
            'persons'        => [],
            'access'         => [],
            'signatory'      => null,
        ])->render();
    }

    public function test_scope_of_works_renders_when_populated(): void
    {
        $html = $this->renderScope('Install Neat Bar Pro below display in the Board Room.');
        $this->assertStringContainsString('Install Neat Bar Pro below display in the Board Room.', $html);
        $this->assertStringNotContainsString('Scope of works not generated', $html);
        $this->assertStringNotContainsString('AV installation works as per quotation', $html);
    }

    public function test_notice_renders_when_scope_empty(): void
    {
        $html = $this->renderScope('');
        $this->assertStringContainsString('Scope of works not generated', $html);
        $this->assertStringContainsString('color:#CC0000', $html);
        $this->assertStringNotContainsString('AV installation works as per quotation', $html);
        $this->assertStringNotContainsString('should not appear', $html);
    }
}
```
