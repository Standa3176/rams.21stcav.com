Also add a new JS function `generateWorksOverviewFromScope()` (used by the btn-gen-overview button) that calls the same AJAX endpoint as generateScopeOfWorks() but only populates the works_overview field (useful if the user wants to regenerate overview without replacing scope):
```javascript
function generateWorksOverviewFromScope() {
    // Calls the same endpoint as the scope-of-works generate.
    // Endpoint returns both fields; this function only writes works_overview.
    const btn = document.getElementById('btn-gen-overview');
    const field = document.getElementById('works-overview-field');
    if (!btn || !field) return;

    const originalText = btn.innerHTML;
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span> Generating&hellip;';

    fetch(btn.closest('form')
            ? btn.closest('[data-scope-url]')?.dataset?.scopeUrl ?? ''
            : '', {
        // Use the same route as the scope-of-works button.
        // Find the URL from the existing scope generate button on the page.
    });

    // Simpler approach: reuse generateScopeOfWorks endpoint.
    // Find the existing scope generate button's click handler endpoint.
    // Since both buttons call the same AJAX URL, extract the URL from the existing btn-gen-scope button.
    const scopeBtn = document.getElementById('btn-gen-scope');
    const scopeUrl = scopeBtn ? scopeBtn.dataset.url : null;

    if (!scopeUrl) {
        btn.disabled = false;
        btn.innerHTML = originalText;
        alert('Cannot find scope endpoint. Please use the Scope of Works Generate button instead.');
        return;
    }

    fetch(scopeUrl, {
        method: 'POST',
        headers: {
            'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content,
            'Accept': 'application/json',
            'Content-Type': 'application/json',
        },
        body: JSON.stringify({}),
    })
    .then(r => r.json())
    .then(data => {
        if (data.works_overview) {
            field.value = data.works_overview;
        } else if (data.error) {
            alert('Could not generate works overview: ' + data.error);
        }
    })
    .catch(() => alert('Request failed. Please try again.'))
    .finally(() => {
        btn.disabled = false;
        btn.innerHTML = originalText;
    });
}
```

IMPORTANT: To make the above work, ensure the existing scope generate button (`btn-gen-scope`) has a `data-url` attribute exposing the AJAX endpoint URL. If it calls `generateScopeOfWorks()` via onclick and the URL is defined as a JS variable, adapt accordingly — use whatever URL reference pattern is already established.
